```dart
class LockMeasurementScreen extends ConsumerWidget {

            ),
            const SizedBox(height: 20),

              onPressed: () async {
                await notifier.measureLock();

                  Navigator.pushNamed(context, '/result');
                }
              },
              child: const Text('Измерить замок'),
            ),
          ],
        ),
      ),

  }
}
```

## Backend - FastAPI

```python
from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from .schemas import LockMeasurementCreate, LockMeasurementResponse
from .services import LockService

app = FastAPI()

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_methods=["*"],
    allow_headers=["*"],
)

@app.post("/locks/measure", response_model=LockMeasurementResponse)
async def measure_lock(data: LockMeasurementCreate):
    lock_service = LockService()
    result = await lock_service.create_measurement(data)
    if not result:
        raise HTTPException(status_code=500, detail="Measurement failed")
    return result
```

## Backend - Service

```python
from .database import async_session
from .models import LockMeasurement

class LockService:
    async def create_measurement(self, data: LockMeasurementCreate) -> LockMeasurementResponse:
        measured_length = 0.0
        async with async_session() as session:
            lock = LockMeasurement(image_url=data.image_url, length_mm=measured_length)
            session.add(lock)
            await session.commit()
            await session.refresh(lock)
            return LockMeasurementResponse(
                id=lock.id,
                measured_length_mm=lock.length_mm,
                status="ok"
            )
```

## CV - Measurement

```python
import cv2
import numpy as np

def detect_lock_and_measure(image_path: str) -> float:
    image = cv2.imread(image_path)
    if image is None:
        raise FileNotFoundError(f"Image not found: {image_path}")

    # YOLO detection
    net = cv2.dnn.readNetFromDarknet('yolov5.cfg', 'yolov5.weights')
    net.setInput(cv2.dnn.blobFromImage(image, 1/255.0, (640, 640), swapRB=True, crop=False))
    outputs = net.forward(net.getUnconnectedOutLayersNames())

    # Find bounding boxes
    boxes = []
    for output in outputs:
        for detection in output:
            scores = detection[5:]
            class_id = np.argmax(scores)
            confidence = scores[class_id]
            if class_id == 0 and confidence > 0.5:
                box = detection[0:4] * np.array([image.shape[1], image.shape[0], image.shape[1], image.shape[0]])
                (centerX, centerY, width, height) = box.astype("int")
                x = int(centerX - width/2)
                y = int(centerY - height/2)
                boxes.append((x, y, int(width), int(height)))

    if not boxes:
        raise ValueError("Lock not detected")
    
    x, y, w, h = max(boxes, key=lambda b: b[2]*b[3])
    pixels_per_mm = calibrate_pixels_per_mm(image)
    length_pixels = max(w, h)
    length_mm = length_pixels / pixels_per_mm
    return length_mm

def calibrate_pixels_per_mm(image) -> float:
    known_length_mm = 10.0
    marker_pixels = 100.0
    return marker_pixels / known_length_mm
```
